	The presence of complicated topography can have significant influence on the atmospheric state, leading to complicated \textbf{mesoscale phenomena.} Our main goal is to capture \textbf{internal gravity waves}, which influence

\section{Mathematical description}
\label{sec:MATH}
\begin{frame}{Governing equations}

	\begin{itemize}
		\item air is modelled as a dry inviscid ideal gas
		\item inside $\Omega \subset \mathbb{R}^2$ a rectangular domain with curved bottom boundary the evolution is described as an initial-boundary value problem for the \textbf{compressible isentropic Euler equations}
	\end{itemize}
	\begin{math-block}{Governing equations}
		\begin{equations}[columns,*]
			\rho \dv{\vb{v}}{t} &= -\grad p + \rho \vb{g},                && \eqnote{momentum balance} \\
			\dv{\rho}{t} + \rho\left(\divergence{\vb{v}}\right) &= 0,     && \eqnote{mass conservation} \\
			\dv{\theta}{t} &= 0,                                      && \eqnote{isentropic evolution} \\
			p\left(\rho, T\right) &= \rho R T.                     && \eqnote{thermal equation of state of an ideal gas}
		\end{equations}
	\end{math-block}
\end{frame}

\begin{frame}{Initial conditions}
	The initial conditions are given by an \textbf{isothermal hydrostatic balance} at $T_0 = \SI{250}{\kelvin}$
	\begin{math-block}{Initial conditions}
		\begin{equations}[columns,*]
			\vb{v}\left(0, \vb{x}\right) &= \vb{0},\\
			\rho\left(0, \vb{x}\right) &= \bar{\rho}(z) \coloneq \rho_0 \exp\left(-z \frac{g}{R T_0}\right), \\
			\theta\left(0, \vb{x}\right) &= \bar{\theta}(z) \coloneq T_0 \exp\left(z\frac{R}{c_p}\frac{g}{R T_0}\right),
		\end{equations}
	\end{math-block}
	and from the state equations it follows for the pressure

	\begin{equations}[single,*]
		p\left(0, \vb{x}\right) = \bar{p}(z) \coloneq \rho_0 R T_0 \exp\left(-z \frac{g}{R T_0}\right).
	\end{equations}

\end{frame}

\section{Numerical solution}
\label{sec:NUMERICS}
\begin{frame}{Smoothed Particle Hydrodynamics}
	The governing equations are solved numericaly using the \textbf{Smoothed Particle Hydrodynamics (SPH)} method
	\begin{itemize}
		\item the fluid is partitioned into \emph{abstract material points}, so called \textbf{particles}
		\item particle $a$ possess its $\vb{v}_a, \rho_a, \theta_a, p_a$ and interact \textbf{locally}
		\item \emph{PDE's} for the continuum become \textbf{ODE's} for the particles
		\item main tool is the \textbf{interpolation kernel} $w_h\left(\vb{x}\right)$
	\end{itemize}

	\begin{equations}[columns,*]
		\Phi\qty(t,\vb{x}) = \Phi\qty(t,\vb{x}) \star \delta\qty(\vb{x}) &=\int_{\Omega}\Phi\qty(t,\vb{x}') \delta\qty(\vb{x}-\vb{x}')\dd{\vb{x}'} \Rightarrow && \eqnote{convolution identity} \\
		[\Phi]^C_{w_h}\qty(t,\vb{x}) \coloneq \Phi\qty(t,\vb{x}) \star w_h\qty(\vb{x}) &= \int_{\Omega}\Phi\qty(t,\vb{x}')w_h\qty(\vb{x}-\vb{x}')\dd{\vb{x}'}, && \eqnote{continuous interpolation} \\
		\Phi_a(t) \equiv [\Phi]_{w_h}^D(t,\vb{x}_a) &\coloneq \sum_b \Phi_a w_{ab} V_b, && \eqnote{discrete interpolation}\\
		\Phi(t,\vb{x}) &\approx [\Phi]_{w_h}^C\qty(t,\vb{x}) \approx \Phi_a(t). && \eqnote{particle approximation}
	\end{equations}
\end{frame}

\begin{frame}{Different formulations}
	\begin{columns}[T,onlytextwidth]
		\column{0.4\textwidth}
		\centering
		\begin{math-block}{Classical SPH}
			\begin{equations}[columns,*]
				\dv{\rho_a}{t} &= \sum_b m_b w'_{ab} \vb{v}_{ab}\vdot \vb{e}_{ab}, \\
				\dv{\vb{v}_a}{t} &= -\sum_b\qty(\frac{p_a^{2}}{\rho_a}+\frac{p_b^{2}}{\rho_b})w'_{ab}\vb{e}_{ab} + \vb{g}, \\
				\theta &= T_a \left(p_0/p_a\right)^{\frac{R}{c_p}} \approx \, \text{const} \,,
			\end{equations}
		\end{math-block}
		\begin{itemize}
			\item poor conservation properties with \link[h]{adaptive $h$},
			\item $\theta$ has to be computed at extra cost.
		\end{itemize}

		\column{0.55\textwidth}
		\centering
		\begin{math-block}{$p-\theta$ formulation}
			\begin{equations}[columns,*]
				P_a &= \left(\sum_{b} \frac{m_b}{\tilde{\theta_0}} \theta_b w_{ab}\left(\link[h]{h_a}\right)\right)^{\gamma},\\
				\dv{\vb{v}_a}{t} &= - \sum_{b} m_b \frac{\theta_a \theta_b}{\tilde{\theta}_0^{2}}\left(\frac{\link[f]{f_{a}}P_a}{P_a^{2/\gamma}} w'_{ab}\left(\link[h]{h_a}\right) + \frac{\link[f]{f_{b}}P_b}{P_b^{2/\gamma}} w'_{ab}\left(\link[h]{h_b}\right)\right)\vb{e}_{ab} + \vb{g},\\
				\theta_a &= \, \text{const} \,
			\end{equations}
		\end{math-block}
		\begin{itemize}
			\item naturally guarantees conservation with \link[h]{adaptive $h$} via the $\link[f]{\grad h}$ \link[f]{terms} $\link[f]{f_{a}}$,
			\item uses the primary variable $\theta$ directly.
		\end{itemize}
	\end{columns}
\end{frame}

\begin{frame}{Well-balanced formulations}
	%TODO: Add these
	Hydrostatic equilibrium $\bar{\rho}, \bar{p}$ solves

	\begin{equations}[single,*]
		\cancel{\bar{\rho} \dv{\vb{v}}{t}} + \grad \bar{p} - \bar{\rho} \vb{g} = \vb{0},
	\end{equations}

however, due to discretization errors, general numerical approximations $\bar{\rho_h}, \bar{p_h}, \grad_h$ \alert{do not solve the hydrostatic balance on the discrete level}:

\begin{equations}[single,*]
	\grad_h \bar{p_h} - \bar{\rho_h} \vb{g} \neq \vb{0} \Rightarrow \, \text{\textbf{spurious force!}} \,
\end{equations}

A numerical scheme that preserves exactly the \emph{analytical stationary solutions} on the \textbf{discrete level} is said to be \textbf{well-balanced.}

\begin{math-block}{Well-balancing: substituting for gravity}
\begin{equations}[single,*]
	\vb{g} = \frac{1}{\bar{\rho}} \grad \bar{p} \Rightarrow \dv{\vb{v}}{t} = - \frac{1}{\rho} \grad p + \frac{1}{\bar{\rho}} \grad \bar{p}
\end{equations}
\end{math-block}

The gradients (forces) on the right-hand side are discretized identically $\Rightarrow$ \textbf{identical truncation errors cancel out}.

\end{frame}

\begin{frame}{Summary and future plans}
	\begin{columns}[T,onlytextwidth]
		\column{0.4\textwidth}
		\centering
		\textbf{Summary}
		\begin{itemize}
			\setlength{\itemsep}{1.5em} 
			\item flows around a topography induce internal gravity waves
			\item IGWs play a significant role in mesoscale atmospheric dynamics
			\item large scale hydrostatic balance is problematic for classical SPH 
			\item well-balanced schemes must to be used for non-stationary flows
		\end{itemize}

		\centering
		\textbf{Plans}
		\begin{itemize}
